async function enviarImagem(file) { const formData = new FormData(); formData.append("file", file); formData.append("upload_preset", "ONCA_UPLOAD"); const response = await fetch( "https://api.cloudinary.com/v1_1/difiv4ogr/image/upload", { method: "POST", body: formData } ); const data = await response.json(); if (data.secure_url) { return data.secure_url; } else { console.log(data); throw new Error("Erro no upload"); } }
